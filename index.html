    import discord
from discord.ext import commands
from discord.ui import Button, View, Modal, TextInput

TOKEN = "ТВІЙ_ТОКЕН_БОТА"
ORDERS_CHANNEL_ID = 123456789012345678

intents = discord.Intents.default()
intents.message_content = True
intents.members = True
bot = commands.Bot(command_prefix="!", intents=intents)

class DeliverOrderModal(Modal, title="Видача файлу / сайту замовнику"):
    client_tag = TextInput(label="Discord нік клієнта (без @)", placeholder="cyber_king")
    result_link = TextInput(label="Посилання на файл або архів сайту", style=discord.TextStyle.paragraph, placeholder="https://mega.nz/... або https://github.com/...")

    async def on_submit(self, interaction: discord.Interaction):
        guild = interaction.guild
        member = discord.utils.find(
            lambda m: m.name.lower() == self.client_tag.value.strip().lower().lstrip("@"),
            guild.members
        )
        if member:
            try:
                await member.send(
                    f"🎉 **Твоє замовлення від NeoWeb готове!**\n"
                    f"📦 **Файл / архів сайту:** {self.result_link.value}\n"
                    f"Дякуємо за покупку! Якщо будуть питання — звертайся."
                )
                await interaction.response.send_message(f"✅ Файл сайту успішно надіслано в особисті повідомлення користувачу {member.mention}!", ephemeral=True)
            except discord.Forbidden:
                await interaction.response.send_message(f"⚠️ У {member.mention} закриті особисті повідомлення (ЛС).", ephemeral=True)
        else:
            await interaction.response.send_message(f"❌ Користувача `{self.client_tag.value}` не знайдено на сервері (він має бути учасником сервера).", ephemeral=True)

class OrderDecisionView(View):
    def __init__(self):
        super().__init__(timeout=None)

    @discord.ui.button(label="Прийняти", style=discord.ButtonStyle.success, emoji="✅")
    async def accept_btn(self, interaction: discord.Interaction, button: Button):
        await interaction.response.send_message(
            f"✅ Заявку **ПРИЙНЯТО** адміністратором {interaction.user.mention}! Тепер натисніть кнопку **📦 Видати сайт**, щоб надіслати файл замовнику.",
            ephemeral=False
        )

    @discord.ui.button(label="Відхилити", style=discord.ButtonStyle.danger, emoji="❌")
    async def reject_btn(self, interaction: discord.Interaction, button: Button):
        await interaction.response.send_message(
            f"❌ Заявку **ВІДХИЛЕНО** адміністратором {interaction.user.mention}.",
            ephemeral=False
        )
        for item in self.children:
            item.disabled = True
        await interaction.message.edit(view=self)

    @discord.ui.button(label="Видати сайт", style=discord.ButtonStyle.primary, emoji="📦")
    async def deliver_btn(self, interaction: discord.Interaction, button: Button):
        await interaction.response.send_modal(DeliverOrderModal())

@bot.event
async def on_message(message):
    if message.channel.id == ORDERS_CHANNEL_ID and message.webhook_id:
        await message.reply("Керування заявкою та видача сайту:", view=OrderDecisionView())
    await bot.process_commands(message)

bot.run(TOKEN)
